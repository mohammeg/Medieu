    function matchesSearch(d, q){ if(!q) return true; const text = [d.name,d.address,d.speciality,d.university,d.college].join(' ').toLowerCase(); return text.includes(q.toLowerCase()); }
    function sortList(list, sort){ const out = [...list]; out.sort((a,b) => { let fa='', fb=''; if(sort.startsWith('name')){fa=a.name;fb=b.name;} else if(sort.startsWith('speciality')){fa=a.speciality;fb=b.speciality;} else if(sort.startsWith('university')){fa=a.university;fb=b.university;} else if(sort.startsWith('college')){fa=a.college;fb=b.college;} else {fa=a.address;fb=b.address;} let cmp = fa.localeCompare(fb); return sort==='name-desc' ? -cmp : cmp; }); return out; }
    function currentList(){ return sortList(doctors.filter(d => matchesSearch(d, state.search) && matchesFilter(state.university, d.university) && matchesFilter(state.college, d.college) && matchesFilter(state.speciality, d.speciality)), state.sort); }

    function renderActive(){ const items=[]; if(state.search) items.push(`Search: "${state.search}"`); state.university.forEach(v => items.push(`University: ${v}`)); state.college.forEach(v => items.push(`College: ${v}`)); state.speciality.forEach(v => items.push(`Speciality: ${v}`)); activeFilters.innerHTML = items.map(i => `<span class="pill">${i}</span>`).join(''); }

    function renderCards(){ const list = currentList(); grid.innerHTML=''; countBox.textContent = `${list.length} doctor${list.length===1?'':'s'} found`; renderActive(); empty.style.display = list.length ? 'none' : 'block';
      list.forEach(d => {
        const card = document.createElement('div'); card.className = 'card'; card.tabIndex = 0;
        card.innerHTML = `<div class="photo"><img src="${d.photo}" alt="${d.name}"><div class="overlay"><div class="name">${d.name}</div><div class="meta">${d.speciality}</div><div class="mini">${d.university}</div></div></div>`;
        card.onclick = () => openDetails(d.id);
        card.onkeydown = e => { if(e.key==='Enter' || e.key===' '){ e.preventDefault(); openDetails(d.id);} };
        grid.appendChild(card);
      });
    }

    function openDetails(id){ const d = doctors.find(x => x.id === id); if(!d) return; detailsView.innerHTML = `
      <button class="back" id="backBtn">← Back to Directory</button>
      <div class="details-grid">
        <div class="profile">
          <div class="profile-img"><img src="${d.photo}" alt="${d.name}"></div>
          <div class="profile-info">
            <h2>${d.name}</h2>
            <div class="rating">⭐ ${d.rating} Rating</div>
            <div class="cta"><button class="cta1">Send Message</button><button class="cta2">Book Appointment</button></div>
          </div>
        </div>
        <div class="panel">
          <div class="title-row"><h1>${d.name}</h1><div class="verify">Verified Specialist</div></div>
          <div class="info-grid">
            <div class="info"><span class="label">Speciality</span><span class="value">${d.speciality}</span></div>
            <div class="info"><span class="label">University</span><span class="value">${d.university}</span></div>
            <div class="info"><span class="label">College</span><span class="value">${d.college}</span></div>
            <div class="info"><span class="label">Address</span><span class="value">${d.address}</span></div>
            <div class="info"><span class="label">Birthday</span><span class="value">${d.birthday}</span></div>
            <div class="info"><span class="label">Graduation Year</span><span class="value">${d.graduationYear}</span></div>
            <div class="info"><span class="label">Experience</span><span class="value">${d.experience}</span></div>
            <div class="info"><span class="label">Phone</span><span class="value">${d.phone}</span></div>
            <div class="info"><span class="label">Email</span><span class="value">${d.email}</span></div>
          </div>
          <div><h3>About</h3><div class="copy">${d.about}</div></div>
          <div><h3>Skills</h3><div class="chips">${d.skills.map(s => `<span class="chip">${s}</span>`).join('')}</div></div>
        </div>
      </div>
      <div class="gallery-block">
        <div class="gallery-head"><h3>Extra Photos</h3><div class="verify">Gallery</div></div>
        <div class="gallery">${d.gallery.map(g => `<img src="${g}" alt="${d.name} extra photo">`).join('')}</div>
      </div>`;
      directoryView.classList.add('hidden'); detailsView.classList.remove('hidden'); window.scrollTo(0,0);
      document.getElementById('backBtn').onclick = () => { detailsView.classList.add('hidden'); directoryView.classList.remove('hidden'); window.scrollTo(0,0); };
    }

    searchInput.addEventListener('input', e => { state.search = e.target.value.trim(); renderCards(); });
    sortSelect.addEventListener('change', e => { state.sort = e.target.value; renderCards(); });
    document.getElementById('applyBtn').onclick = () => { state.university = checked('university'); state.college = checked('college'); state.speciality = checked('speciality'); renderCards(); closeSidebar(); };
    document.getElementById('resetBtn').onclick = () => { state.university=[]; state.college=[]; state.speciality=[]; setChecked('university',[]); setChecked('college',[]); setChecked('speciality',[]); renderCards(); };
    document.addEventListener('keydown', e => { if(e.key==='Escape') closeSidebar(); });

    renderCards();
  </script>
